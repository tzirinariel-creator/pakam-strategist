/**
 * Which sitting to recommend. Default is ALWAYS Moed A (the last grade counts —
 * B is the safety net, and sitting A keeps B in reserve). Recommend B only when
 * A is tight (<3 days from another chosen exam) and B is not. Null = no dates.
 */
export function recommendMoed(
  course: { examDateA: Date | null; examDateB: Date | null },
  otherChosenDates: Date[],
): "A" | "B" | null {
  if (!course.examDateA && !course.examDateB) return null;
  if (!course.examDateA) return "B";
  if (!course.examDateB) return "A";
  const tight = (d: Date) =>
    otherChosenDates.some((o) => Math.abs(startOfDay(o).getTime() - startOfDay(d).getTime()) < 3 * 86400000);
  return tight(course.examDateA) && !tight(course.examDateB) ? "B" : "A";
}
```

// StudyTask rows. Pure + unit-tested; returns null when there is no upcoming plan.
import { planFromStudyTasks, type StudyTaskLike } from "@/lib/plan-from-tasks";

export function buildExamPeriodBlock(tasks: StudyTaskLike[], now: Date = new Date()): string | null {
  const plan = planFromStudyTasks(tasks, new Map(), now);
  if (plan.exams.length === 0) return null;

  const fmt = (d: Date) => `${d.getDate()}.${d.getMonth() + 1}`;

    .slice().sort((a, b) => a.examDate.getTime() - b.examDate.getTime())
    .slice(0, 10)
    .map((e) => `  • ${e.courseName} — ${fmt(e.examDate)} (מועד ${e.moed === "B" ? "ב׳" : "א׳"}), ${e.totalHours} שעות לימוד מתוכננות`)
    .join("\n");

  // Heaviest planned day (sum of session hours per local day).
  const byDay = new Map<string, number>();
  for (const s of plan.sessions) {
    const k = `${s.date.getDate()}.${s.date.getMonth() + 1}`;
    byDay.set(k, (byDay.get(k) ?? 0) + s.hours);
  }
  const peak = [...byDay.entries()].sort((a, b) => b[1] - a[1])[0];

  return `

## תקופת המבחנים של הסטודנט (מתוך תוכנית-הלימוד השמורה באפליקציה — מקור-אמת, אל תמציא תאריכים):
${exams}
  סה"כ מפגשי-לימוד מתוכננים: ${plan.sessions.length}${peak ? ` · היום העמוס ביותר: ${peak[0]} (${peak[1]} שעות)` : ""}

כלל לשאלות תכנון-מבחנים (גובר על "אל תשאל הבהרות" לנושא הזה בלבד): פתח במשפט אחד שמשקף את המצב מהנתונים שלמעלה, שאל שאלה ממוקדת אחת (יום חסום? עבודה? קורס שמדאיג?) — ורק אחרי התשובה תן המלצה קונקרטית אחת. אם אין כאן תוכנית — הפנה למסך "מבחנים" לבנות אחת.`;
}
```
(משתמשים ב-`planFromStudyTasks` הקיים — כבר מסנן עבר-בלבד, מפענח `[auto] Xh` ומועד-ב'; אפס לוגיקה כפולה. `cleanCourseName` מחלץ שם מהכותרת גם בלי מפת-קודים.)

## סיכונים ונקודות-שבירה

1. **שינוי ה-`backwards` הוא גלובלי.** הוא מתקן משפחת-באגים שלמה, אבל אם אי-שם רכיב הסתמך (בטעות) על ה-transform השיורי — הוא ישתנה. הסבירות אפסית (מצב-הסיום זהה), אבל זו הסיבה לבדיקת-המעבר הוויזואלית על כל הדפים בסעיף הידני. אם משהו נשבר — לתקן שם נקודתית, לא להחזיר `both`.
2. **תשובות קטועות ישנות בקאש.** `answer-cache` (localStorage, מפתח `pk-ai-cache-v1`) עשוי להחזיק תשובות שנחתכו לפני התיקון. הוא מתאפס עם כל שינוי-תוכנית, אז זה מתפוגג לבד; אם רוצים ניקוי מיידי — להעלות את `NS` ל-`pk-ai-cache-v2` (`src/lib/ai/answer-cache.ts:11`). לבצע את העלאת-הגרסה — שורה אחת, בטוחה.
3. **`stop_reason` בנתיב Claude:** ב-SDK של Anthropic הערך הוא `"max_tokens"` על `finalMessage()`. אם טיפוס ה-SDK מחמיר — להשוות כמחרוזת (`final.stop_reason === "max_tokens"`), לא להוסיף casting מעבר לזה.
4. **שאילתת studyTask בכל הודעת-צ'אט** — עוד קווארי אחד קטן פר-הודעה (אינדקס על userId קיים דרך ה-relation). זניח ב-24 משתמשים; אם אי-פעם יכאב — לקאש ב-context שכבר נשמר לסשן.
5. **המלצת-מועד דינמית:** שינוי מועד בקורס אחד יכול להזיז את תג-ה"מומלץ" בקורס אחר (כי הצפיפות תלוית-בחירה). זו התנהגות נכונה (המלצה חיה), אבל אם זה מבלבל בבדיקה הידנית — להקפיא את ההמלצה לתאריכי מועד-א' של האחרים בלבד (שינוי של שורה ב-`otherChosenDates`). ברירת-המחדל: להשאיר דינמי.
6. **`DropdownMenuItem` ו-download:** `onSelect` של Radix רץ אחרי סגירת-התפריט; יצירת `<a download>` ב-`exportIcs`/`downloadGanttCsv` לא תלויה בפוקוס — אין בעיה ידועה, אבל אם דפדפן חוסם את ההורדה, להוסיף `event.preventDefault()` ב-onSelect ולקרוא לפונקציה ישירות.
7. **גבול-אמון:** בלוק-המבחנים נבנה בשרת מנתוני-DB (לא מ-hint של הלקוח) — אין וקטור-הזרקה חדש; `isSafeDeterministicHint` ממשיך לחול רק על ה-hint. אסור "לקצר" ולהעביר את התוכנית דרך ה-deterministicHint מהלקוח — זה עוקף את הסינון בכוונה תחילה.

2. **באג מובנה בצעד 4.4: הכפתור "לחשוב על זה עם המלך" ייכנס גם ל-STATE A.** `recsCard` הוא sub-render משותף שמרונדר גם במצב-תצוגה-מקדימה (שורה 364, `previewRecs`) — לפני שקיימת תוכנית שמורה. שם `studyTask` ריק ⇒ `buildExamPeriodBlock` מחזיר null ⇒ לפי כלל-השיחה החדש המלך יענה "היכנס למסך 'מבחנים' לבנות תוכנית" — למשתמש שנמצא במסך המבחנים ומסתכל על תוכנית-פריוויו. לולאה אבסורדית. **תיקון:** לרנדר את הכפתור רק כש-`hasPlan` (להעביר דגל ל-`recsCard`, או להוסיף את הכפתור רק בקריאה של STATE B בשורה 344).

3. **החלטה מוסתרת בסעיף-הסיכונים: העלאת גרסת ה-cache.** סיכון #2 פותח ב"זה מתפוגג לבד; **אם רוצים** ניקוי מיידי — להעלות ל-pk-ai-cache-v2" ואז מסיים "**לבצע** את העלאת-הגרסה". זה גם סתירה פנימית וגם ממוקם מחוץ למפרט-הביצוע — Opus שמבצע את צעדים 1-5 לא יגיע לזה, ואין לזה סעיף-בדיקה. **תיקון:** להעביר כצעד 1.3 מפורש ("`answer-cache.ts:11`: `NS = "pk-ai-cache-v2"`") או למחוק את משפט ה"לבצע" ולהשאיר את ההתפוגגות הטבעית. אחד מהשניים, בגוף המפרט.

4. **הצהרת-חסר בצעד 2.1: "מעלים את ה-transform התמידי מכל האפליקציה".** לא מדויק — `.animate-rise` (`globals.css:653`) גם הוא `both` עם keyframe של `translateY`, כלומר אותה מלכודת בדיוק. כרגע המחלקה מתה (אפס שימושים ב-src), אז אין באג חי, אבל הטענה כלשונה שגויה והמוקש נשאר למי שישתמש בה מחר. **תיקון:** לכלול את `animate-rise` בסוויפ (עוד `both`→`backwards` אחד) ולדייק את הניסוח.

5. **גזירת scope בלי סטטוס-בעלים: "ייצוא-אקסל ברמת-גאנט אמיתית".** זו דרישה מפורשת בהערות-הסיכום (הערות-סיכום-אריאל.md:35), והמפרט דוחה אותה חד-צדדית ("מה לא עושים" #2). הרציונל טוב (ה-CSV הוא תיקון מכוון לקובץ שאקסל דחה), אבל תחת משטר "אפס נפילות — כל הערה מקבלת סטטוס" זו החלטת-בעלים, לא החלטת-מפרט. **תיקון:** להוסיף שורה מפורשת: לרשום את הדחייה בקובץ-הבקרה (docs/משימות-בקרה.md) כ"ממתין לאישור אריאל / נדחה-לאחרי-השקה", כדי שההערה לא תיפול בין הכיסאות.

6. **תג "מומלץ" ב-10px מותח את קו-העיצוב מעבר למה שהציטוט מרשה.** שורה 41 מתירה מ-10px "תוויות-מיקרו **באותיות בודדות**, uppercase, בלבד" — "מומלץ" הוא מילה שלמה. יש אמנם תקדימים בקובץ עצמו (`ש״ס` ב-text-[10px] בשורה 249, ואפילו text-[9px] בשורה 490 שמפר את שורה 64 בקו-העיצובי — הפרה קיימת, לא של המפרט), אבל המפרט מציג את הציטוט כהיתר מלא. **תיקון:** `text-[11px]` לתג (עדיין מיקרו, לא שנוי במחלוקת), או לנסח בכנות "בהתאם לתקדים הקיים בקובץ".

7. **גילוי-החיתוך יכול להחמיץ את הצ'אנק האחרון בקצה:** לולאת הפרסינג ב-`streamGemini` (שורות 164-187) זורקת את שארית ה-buffer אחרי `done` — שורת `data:` אחרונה שלא נחתמה ב-newline לעולם לא מפורסרת, כולל ה-`finishReason` שעליו הפתרון נשען. ב-SSE תקין אירוע מסתיים ב-`\n\n` אז זה נדיר, אבל מנגנון-הכנות לא צריך להיות תלוי בזה. **תיקון:** שורת flush אחרי הלולאה (`if (buffer.trim().startsWith("data:")) {...parse...}`) — שלוש שורות, מחסן גם את הטקסט וגם את הדגל.

8. **קלים:** (א) המפרט לא נותן אף פעם את הנתיב המלא של הקובץ המרכזי — `src/app/[locale]/(protected)/exam-planner/exam-planner-content.tsx`; Opus ימצא אותו ב-glob, אבל מפרט שמצטט file:line בכל שורה צריך לתת נתיב פעם אחת. (ב) `quickAdd` יוצר כותרת עברית קשיחה (`לימוד: X`) גם בלוקאל אנגלי — עקבי עם המשימות האוטומטיות (שגם הן עבריות), אז נסבל, אבל שווה הערת-מודעות. (ג) `pushWholeDay` עם `Promise.all` של N `mutateAsync` יורה N אינבלידציות ⇒ N refetches; עדיף `invalidate` אחד בסוף (או לפחות לציין שזה מקובל ב-N קטן).

## מה חסר

שלושה דברים שהמפרט לא מכסה: **(1) הנתיב Claude-BYOK בבדיקות** — יש בדיקת-יחידה לגילוי-חיתוך של Gemini (סעיף-בדיקות 4) אבל אף בדיקה לענף `final.stop_reason === "max_tokens"` של Claude; ערך-ההחזר החדש של `produce` נבדק רק בצד אחד. **(2) אינטראקציית cache↔"המשך"** — תשובה שנחתכה ונכנסה ל-answer-cache תוגש בעתיד מהקאש בלי לעבור בשרת, כלומר בלי להירשם להיסטוריית-הסשן; ה"המשך" שהשורה החדשה מזמינה יגיע אז למלך שלא ראה את התשובה. ההסתברות נמוכה (עם תקרת 2000 חיתוך נהיה נדיר), אבל שווה משפט אחד: לא לקאש תשובות עם `truncated=true` (תנאי אחד ב-`floating-assistant`) — או לקבל את הסיכון בכתב. **(3) שקיפות-עלות ל-BYOK** — העלאת `CHAT_MAX_TOKENS` ל-2000 מעלה את תקרת-העלות פר-תשובה למשתמשי Claude-בתשלום (Gemini חינמי ממילא); זה שינוי נכון, אבל סעיף-הסיכונים מכסה את שאילתת-ה-DB הזניחה ולא את זה. מעבר לכך — סדר-הביצוע, סגירות-ההחלטות (למעט ליקוי 3), ההלימה לקו-העיצובי (Radix-בפורטל הוא בדיוק הקאנון של שורה 35) וההלימה לתוכנית-הסיום (אין נושא VI שם; "באגי המלך #36" משובצים ביום-4 — אין סתירה) — תקינים. אחרי תיקון ליקויים 1-3 זה מפרט שאפשר למסור ל-Opus 4.8 בביטחון.
